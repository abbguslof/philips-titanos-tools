/usr/bin/id > /media/sda1/d/diag3.txt 2>&1
/bin/cat /sys/fs/selinux/enforce >> /media/sda1/d/diag3.txt 2>&1
/bin/cat /proc/tty/driver/serial >> /media/sda1/d/diag3.txt 2>&1
/sbin/ip -4 addr >> /media/sda1/d/diag3.txt 2>&1
/bin/cp /media/sda1/busybox-armhf /tmp/busybox
/bin/chmod +x /tmp/busybox
/usr/sbin/iptables -I INPUT -p tcp --dport 2323 -j ACCEPT
/usr/sbin/iptables -S INPUT > /media/sda1/d/ipt3.txt 2>&1
/tmp/busybox telnetd -p 2323 -l /bin/sh -f /media/sda1/d/telnetd3.log &
/bin/sleep 2
/bin/netstat -ltn > /media/sda1/d/net3.txt 2>&1
/bin/sh -c 'grep -q ":2323" /media/sda1/d/net3.txt || { echo 0 > /sys/fs/selinux/enforce; /tmp/busybox telnetd -p 2323 -l /bin/sh; /bin/netstat -ltn > /media/sda1/d/net3b.txt; }'

/bin/cat /proc/tty/driver/serial > /media/sda1/d/rx_after.txt
